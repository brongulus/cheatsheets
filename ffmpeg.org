* Example Snippets and Flags
** Downscale:
    -vf scale=-1:720

** 10 bit Video: 
    -pix_fmt yuv420p10le

** AAC Audio:
    -c:a aac -b:a 128k

** Add subtitles:
    #+BEGIN_SRC sh
    ffmpeg -i in.mkv -i sub.srt -c:v copy -c:a copy -c:s copy out.mkv
    #+END_SRC
    
** Merge Audio and Video Streams:
    #+BEGIN_SRC sh
    ffmpeg -i in.mp4 -i audio.mp3 -c:v copy -c:a aac -strict experimental output.mp4 \\
    @pause
    #+END_SRC
    
** Constant Frame Rate:
    -framerate 24000/1001

** MPEG-TS stream to mp4:
    #+BEGIN_SRC sh  
    ffmpeg -y -f mpegts -i in.ts -c copy out.mp4
    #+END_SRC

** Removing Subtitles:  
    #+BEGIN_SRC sh  
    ffmpeg -i video.mkv -vcodec copy -acodec copy -sn video-no-subs.mkv
    #+END_SRC
    
** Stream Detection:  
    #+BEGIN_SRC sh
    # Show the various streams present in a file
    ffprobe -v error -show_entries stream=index,codec_name,codec_type video.mp4 \\
    @pause
    #+END_SRC
    
** Trim Video:
    #+BEGIN_SRC sh
    ffmpeg -ss 00:01:00 -i input.mp4 -to 00:02:00 -c copy output.mp4
    #+END_SRC
    
** Extract Subtitles:
    #+BEGIN_SRC sh
    ffmpeg -i input.mp4 -map 0:s:1 subs.srt \\
    //0:s:0 first subtitle \\
    //0:s:1 second and so on 
    #+END_SRC
    
** Extract Audio:
    #+BEGIN_SRC
    for f in *.mp4;
    do ffmpeg -i "$f" -vn -c:a copy out/"${f%.*}.aac";
    done
    #+END_SRC
    
** Batch Encoding:
    #+BEGIN_SRC sh
    @echo off \\
    for /r %%f in (*.mp4) do ( \\
    ffmpeg -i "%%~nxf" -c:v libx265 -pix_fmt yuv420p10le -crf 26 -c:a aac -b:a 128k "%%~nf.mkv" \\
    ) \\
    @pause
    #+END_SRC
    
** Change Aspect Ratio
    #+BEGIN_SRC sh
    ffmpeg -i input.mp4 -c copy -aspect 4:3 "out.mp4"
    #+END_SRC
    
* Example:
    #+BEGIN_SRC sh
    ffmpeg -i foo1.mp4 -c:v libx265 -r 24000/1001 -pix_fmt yuv420p10le -crf 25 -c:a aac -b:a 128k foo.mkv \\     
    @pause
    #+END_SRC

* iPod album art conversion

#+BEGIN_SRC sh
#!/usr/bin/env bash
shopt -s nullglob

# This doesn't use ffmpeg because it destroys the images and iPod cannot
# handle ffmpeg processed images so need to resort to all this hackery ;-;

# Dependencies
for cmd in eyeD3 magick metaflac; do
    if ! command -v "$cmd" >/dev/null; then
        echo "Error: $cmd is not installed."
        exit 1
    fi
done

for file in *.mp3 *.flac; do
    echo "Processing '$file'..."
    mkdir -p tmp

    # Extract
    if [[ "$file" == *.mp3 ]]; then
        eyeD3 --write-images=tmp "$file" >/dev/null 2>&1
    elif [[ "$file" == *.flac ]]; then
        metaflac --export-picture-to=tmp/FRONT_COVER.jpg "$file" 2>/dev/null
    fi

    if [[ ! -f tmp/FRONT_COVER.jpg ]]; then
        echo "  No embedded thumbnail found, skipping."
        rm -rf tmp
        continue
    fi

    # Convert to baseline JPEG
    magick tmp/FRONT_COVER.jpg -resize 320x320^ -gravity center -extent 320x320 \
           -sampling-factor 4:4:4 -interlace none -colorspace sRGB cover_new.jpg

    # Re-embed
    if [[ "$file" == *.mp3 ]]; then
        eyeD3 --remove-all-images "$file" >/dev/null 2>&1
        eyeD3 --add-image cover_new.jpg:FRONT_COVER "$file" >/dev/null 2>&1
    elif [[ "$file" == *.flac ]]; then
        metaflac --remove --block-type=PICTURE "$file"
        metaflac --import-picture-from=cover_new.jpg "$file"
    fi

    rm -rf tmp cover_new.jpg
    echo "  Done."
done

echo "All files processed."
#+END_SRC
